\section*{Notation}

\begin{table}[H]
	\centering
	\caption{Control Symbols}
	\begin{tabular}{c|c|l}
		Symbol & Code & Description \\
		\hline
		$A$ & \verb|A| & System matrix $\mathbb{R}^{n \times n}$ \\
		$B$ & \verb|B| & Input matrix $\mathbb{R}^{n \times m}$ \\
		$C$ & \verb|C| & Output matrix $\mathbb{R}^{q \times n}$ \\
		$E$ & \verb|E| & Affine term of state space model $\mathbb{R}^{n \times 1}$ \\
        $g$ & \verb|g| & Auxiliary prediction term, see \eqref{equ:2_prediction}\\
		${x}$ & \verb|x| & State vector $\mathbb{R}^{n \times 1}$ \\
		$\xi$ & \verb|xi| & Augmented state vector \\
		${u}$ & \verb|u| & Input vector $\mathbb{R}^{m \times 1}$ (always electric voltage) \\
		$\Delta {u}$ & \verb|Delta_u| & Incremental input vector \\
		$y$ & \verb|y| & Output vector $\mathbb{R}^{q \times 1}$ \\
		$N_P$ & \verb|Np| & Prediction horizon \\
        $D_s$ & \verb|Ds| & Decision steps $=N_P-1$, for delay compensation \\
		$d$ & \verb|d| & Disturbance \\
		$P_x, P_u, P_{\Delta u}, P_E$ & \verb|Px, Pu, PDelta_u, PE| & Prediction matrices for $x, u, \Delta u, E(\omega[k])$ \\
		$P_T$ & \verb|Pt| & Lower block-triangular identity matrix \\
		$\eta$ & \verb|eta| & Slackness variable for soft constraints \\
		$\bm{\rho}$ & \verb|rho| & Scaling/activation vector for scalar $\eta$ \\
	\end{tabular}
\end{table}
